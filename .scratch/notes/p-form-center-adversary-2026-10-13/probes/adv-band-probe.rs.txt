#[test]
#[ignore]
fn adv_probe_all_bands() {
    let stats = l1_stats();
    let n = SEEDS_L1 as f64;
    let s = |f: fn(&MatchStats) -> usize| stats.iter().map(f).sum::<usize>();
    let sf = |f: fn(&MatchStats) -> usize| stats.iter().map(f).sum::<usize>() as f64;
    let (b, bg, bsv, bof) = (s(|x| x.n_box), s(|x| x.n_box_goal), s(|x| x.n_box_saved), s(|x| x.n_box_off));
    let (a, ag, asv, aof) = (s(|x| x.n_arc), s(|x| x.n_arc_goal), s(|x| x.n_arc_saved), s(|x| x.n_arc_off));
    let (f, fg, fsv, fof) = (s(|x| x.n_far), s(|x| x.n_far_goal), s(|x| x.n_far_saved), s(|x| x.n_far_off));
    let regular = b + a + f;
    println!("[ADV] 普通射门/场      {:.3}   band [14,29]", regular as f64 / n);
    println!("[ADV] 禁区内占比        {:.4}   band [0.45,0.65]", b as f64 / regular as f64);
    println!("[ADV] box g/s/o          {:.4}/{:.4}/{:.4}", bg as f64 / b as f64, bsv as f64 / b as f64, bof as f64 / b as f64);
    println!("[ADV] arc g/s/o          {:.4}/{:.4}/{:.4}", ag as f64 / a as f64, asv as f64 / a as f64, aof as f64 / a as f64);
    println!("[ADV] far g/s/o          {:.4}/{:.4}/{:.4}", fg as f64 / f as f64, fsv as f64 / f as f64, fof as f64 / f as f64);
    println!("[ADV] far45              {}   band ==0", s(|x| x.n_shot_far45));
    let (hg, hsv, hoff) = (s(|x| x.n_header_goal), s(|x| x.n_header_saved), s(|x| x.n_header_off));
    println!("[ADV] header g/s/o       {:.4}/{:.4}/{:.4}", hg as f64 / (hg + hsv + hoff) as f64, hsv as f64 / (hg + hsv + hoff) as f64, hoff as f64 / (hg + hsv + hoff) as f64);
    let tackles = s(|x| x.n_tackle);
    println!("[ADV] 抢断/场           {:.3}   band [24,50]", tackles as f64 / n);
    println!("[ADV] 抢断成功率        {:.4}   band [0.42,0.58]", sf(|x| x.n_tackle_success) / tackles as f64);
    println!("[ADV] shot/tackle       {:.4}   band [0.5,1.5]", regular as f64 / tackles as f64);
    let shots_all = regular + hg + hsv + hoff;
    let goals = s(|x| x.n_shot_goal) + hg;
    let saved = s(|x| x.n_shot_saved) + hsv;
    println!("[ADV] 射正率 sot        {:.4}   band [0.28,0.39]", (goals + saved) as f64 / shots_all as f64);
    println!("[ADV] 转化 conv         {:.4}   band [0.08,0.14]", goals as f64 / shots_all as f64);
    println!("[ADV] 禁区内进球占比    {:.4}   band [0.72,0.92]", (s(|x| x.n_box_goal) + hg) as f64 / goals as f64);
    let corners = s(|x| x.n_corner_kick);
    println!("[ADV] 角球/场           {:.3}   band [1,7] (guard)", corners as f64 / n);
    println!("[ADV] 角球单场 max      {}   band <=15", stats.iter().map(|x| x.n_corner_kick).max().unwrap_or(0));
    println!("[ADV] 界外球/场         {:.3}   band [15,60] (guard)", s(|x| x.n_out_sideline) as f64 / n);
    println!("[ADV] 传球成功率        {:.4}   band [0.82,0.90]", s(|x| x.n_pass_success) as f64 / s(|x| x.n_pass) as f64);
    println!("[ADV] 犯规/场           {:.3}   band [16,30]", s(|x| x.n_foul) as f64 / n);
    println!("[ADV] 黄牌/场           {:.3}   band [2,5]", s(|x| x.n_foul_yellow) as f64 / n);
    println!("[ADV] 红牌/场           {:.3}   band <=0.8", s(|x| x.n_foul_red) as f64 / n);

    let ha = ha_stats();
    let hn = SEEDS_HA as f64;
    let gh = ha.iter().map(|x| x.n_goal_home).sum::<usize>();
    let ga = ha.iter().map(|x| x.n_goal_away).sum::<usize>();
    println!("[ADV] 主队进球/场       {:.4}   band [0.85,1.60]", gh as f64 / hn);
    println!("[ADV] 客队进球/场       {:.4}   band >=0.30", ga as f64 / hn);
    println!("[ADV] H/A={:.4}  方向断言 gh>ga: {}", gh as f64 / ga.max(1) as f64, gh > ga);
}
